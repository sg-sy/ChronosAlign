  VALUES ?eventType { wd:Q13406463 }
  ?list wdt:P31 ?eventType .
  
  # Wikipedia page compulsory
  ?wikipediaPage schema:about ?list ;
                 schema:isPartOf <https://en.wikipedia.org/> .

}
